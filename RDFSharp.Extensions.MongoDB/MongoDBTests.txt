/* This is the official test suite of RDFMongoDBStore. It uses a local MongoDB instance (e.g. "docker run -d -p 27017:27017 mongo") */

using RDFSharp.Extensions.MongoDB;
using RDFSharp.Model;
using RDFSharp.Store;
using Xunit;

//Setup
RDFMongoDBStore mongoDBStore = new RDFMongoDBStore("mongodb://127.0.0.1:27017", "rdfsharp");
Assert.NotNull(mongoDBStore);
Assert.Equal("MONGODB", mongoDBStore.StoreType);

//Clear
if (new Random().NextDouble() >= 0.50)
    await mongoDBStore.ClearQuadruplesAsync();
else
    mongoDBStore.ClearQuadruples();

Assert.Equal(0, mongoDBStore.QuadruplesCount);
Assert.Equal(0, await mongoDBStore.QuadruplesCountAsync);

//Load
for (int i = 1; i <= 250; i++)
{
    mongoDBStore.AddQuadruple(new RDFQuadruple(
            new RDFContext(new Uri($"ex:ctx{i}")),
            new RDFResource($"ex:subj{i}"),
            new RDFResource($"ex:pred{i}"),
            new RDFResource($"ex:obj{i}")));
    await mongoDBStore.AddQuadrupleAsync(new RDFQuadruple(
        new RDFContext(new Uri($"ex:ctx{i}")),
        new RDFResource($"ex:subj{i}"),
        new RDFResource($"ex:pred{i}"),
        new RDFPlainLiteral($"{i}")));
}
Assert.Equal(500, mongoDBStore.QuadruplesCount);
Assert.Equal(500, await mongoDBStore.QuadruplesCountAsync);
mongoDBStore.AddQuadruple(new RDFQuadruple(
    new RDFContext(new Uri("ex:ctx200")),
    new RDFResource("ex:subj200"),
    new RDFResource("ex:pred200"),
    new RDFResource("ex:obj200"))); //Already existing
Assert.Equal(500, mongoDBStore.QuadruplesCount);
Assert.Equal(500, await mongoDBStore.QuadruplesCountAsync);
RDFGraph graph1 = new RDFGraph([
    new RDFTriple(new RDFResource("ex:subj251"), new RDFResource("ex:pred251"), new RDFPlainLiteral("251")),
    new RDFTriple(new RDFResource("ex:subj251"), new RDFResource("ex:pred251"), new RDFResource("ex:obj251"))]);
mongoDBStore.MergeGraph(graph1);
Assert.Equal(502, mongoDBStore.QuadruplesCount);
Assert.Equal(502, await mongoDBStore.QuadruplesCountAsync);
RDFGraph graph2 = new RDFGraph([
    new RDFTriple(new RDFResource("ex:subj252"), new RDFResource("ex:pred252"), new RDFPlainLiteral("252")),
    new RDFTriple(new RDFResource("ex:subj252"), new RDFResource("ex:pred252"), new RDFResource("ex:obj252"))]);
await mongoDBStore.MergeGraphAsync(graph2);
Assert.Equal(504, mongoDBStore.QuadruplesCount);
Assert.Equal(504, await mongoDBStore.QuadruplesCountAsync);

//Remove
mongoDBStore.RemoveQuadruple(new RDFQuadruple(
    new RDFContext(),
    new RDFResource("ex:subj252"),
    new RDFResource("ex:pred252"),
    new RDFResource("ex:obj252")));
Assert.Equal(503, mongoDBStore.QuadruplesCount);
Assert.Equal(503, await mongoDBStore.QuadruplesCountAsync);
mongoDBStore.RemoveQuadruple(new RDFQuadruple(
    new RDFContext(),
    new RDFResource("ex:subj254"),
    new RDFResource("ex:pred254"),
    new RDFResource("ex:obj254"))); //unexisting
Assert.Equal(503, mongoDBStore.QuadruplesCount);
Assert.Equal(503, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(c:new RDFContext());
Assert.Equal(500, mongoDBStore.QuadruplesCount);
Assert.Equal(500, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(s:new RDFResource("ex:subj250"));
Assert.Equal(498, mongoDBStore.QuadruplesCount);
Assert.Equal(498, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(p:new RDFResource("ex:pred249"));
Assert.Equal(496, mongoDBStore.QuadruplesCount);
Assert.Equal(496, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(o:new RDFResource("ex:obj248"));
Assert.Equal(495, mongoDBStore.QuadruplesCount);
Assert.Equal(495, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(l:new RDFPlainLiteral("247"));
Assert.Equal(494, mongoDBStore.QuadruplesCount);
Assert.Equal(494, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(c:new RDFContext(new Uri("ex:ctx246")),s:new RDFResource("ex:subj246"));
Assert.Equal(492, mongoDBStore.QuadruplesCount);
Assert.Equal(492, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(c:new RDFContext(new Uri("ex:ctx245")),p:new RDFResource("ex:pred245"));
Assert.Equal(490, mongoDBStore.QuadruplesCount);
Assert.Equal(490, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(c:new RDFContext(new Uri("ex:ctx244")),o:new RDFResource("ex:obj244"));
Assert.Equal(489, mongoDBStore.QuadruplesCount);
Assert.Equal(489, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(c:new RDFContext(new Uri("ex:ctx243")),l:new RDFPlainLiteral("243"));
Assert.Equal(488, mongoDBStore.QuadruplesCount);
Assert.Equal(488, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(c:new RDFContext(new Uri("ex:ctx242")),s:new RDFResource("ex:subj242"),p:new RDFResource("ex:pred242"));
Assert.Equal(486, mongoDBStore.QuadruplesCount);
Assert.Equal(486, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(c:new RDFContext(new Uri("ex:ctx241")),s:new RDFResource("ex:subj241"),o:new RDFResource("ex:obj241"));
Assert.Equal(485, mongoDBStore.QuadruplesCount);
Assert.Equal(485, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(c:new RDFContext(new Uri("ex:ctx240")),s:new RDFResource("ex:subj240"),l:new RDFPlainLiteral("240"));
Assert.Equal(484, mongoDBStore.QuadruplesCount);
Assert.Equal(484, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(c:new RDFContext(new Uri("ex:ctx239")),s:new RDFResource("ex:subj239"),p:new RDFResource("ex:pred239"),o:new RDFResource("ex:obj239"));
Assert.Equal(483, mongoDBStore.QuadruplesCount);
Assert.Equal(483, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(c:new RDFContext(new Uri("ex:ctx238")),s:new RDFResource("ex:subj238"),p:new RDFResource("ex:pred238"),l:new RDFPlainLiteral("238"));
Assert.Equal(482, mongoDBStore.QuadruplesCount);
Assert.Equal(482, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(s:new RDFResource("ex:subj237"),p:new RDFResource("ex:pred237"));
Assert.Equal(480, mongoDBStore.QuadruplesCount);
Assert.Equal(480, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(s:new RDFResource("ex:subj236"),o:new RDFResource("ex:obj236"));
Assert.Equal(479, mongoDBStore.QuadruplesCount);
Assert.Equal(479, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(s:new RDFResource("ex:subj235"),l:new RDFPlainLiteral("235"));
Assert.Equal(478, mongoDBStore.QuadruplesCount);
Assert.Equal(478, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(s:new RDFResource("ex:subj234"),p:new RDFResource("ex:pred234"),o:new RDFResource("ex:obj234"));
Assert.Equal(477, mongoDBStore.QuadruplesCount);
Assert.Equal(477, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(s:new RDFResource("ex:subj233"),p:new RDFResource("ex:pred233"),l:new RDFPlainLiteral("233"));
Assert.Equal(476, mongoDBStore.QuadruplesCount);
Assert.Equal(476, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(p:new RDFResource("ex:pred232"),o:new RDFResource("ex:obj232"));
Assert.Equal(475, mongoDBStore.QuadruplesCount);
Assert.Equal(475, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(p:new RDFResource("ex:pred231"),l:new RDFPlainLiteral("231"));
Assert.Equal(474, mongoDBStore.QuadruplesCount);
Assert.Equal(474, await mongoDBStore.QuadruplesCountAsync);
await mongoDBStore.RemoveQuadruplesAsync(p:new RDFResource("ex:pred231"),l:new RDFPlainLiteral("231")); //no more existing
Assert.Equal(474, mongoDBStore.QuadruplesCount);
Assert.Equal(474, await mongoDBStore.QuadruplesCountAsync);

//Query
Assert.True(mongoDBStore.ContainsQuadruple(new RDFQuadruple(
    new RDFContext(new Uri("ex:ctx200")),
    new RDFResource("ex:subj200"),
    new RDFResource("ex:pred200"),
    new RDFResource("ex:obj200"))));
Assert.True(await mongoDBStore.ContainsQuadrupleAsync(new RDFQuadruple(
    new RDFContext(new Uri("ex:ctx200")),
    new RDFResource("ex:subj200"),
    new RDFResource("ex:pred200"),
    new RDFResource("ex:obj200"))));
Assert.False(mongoDBStore.ContainsQuadruple(new RDFQuadruple(
    new RDFContext(new Uri("ex:ctx2007787")),
    new RDFResource("ex:subj200"),
    new RDFResource("ex:pred200"),
    new RDFResource("ex:obj200"))));
Assert.False(await mongoDBStore.ContainsQuadrupleAsync(new RDFQuadruple(
    new RDFContext(new Uri("ex:ctx2007787")),
    new RDFResource("ex:subj200"),
    new RDFResource("ex:pred200"),
    new RDFResource("ex:obj200"))));
Assert.Equal(2, mongoDBStore.SelectQuadruples(c: new RDFContext(new Uri("ex:ctx200"))).Count);
Assert.Equal(2, mongoDBStore.SelectQuadruples(s: new RDFResource("ex:subj200")).Count);
Assert.Equal(2, mongoDBStore.SelectQuadruples(p: new RDFResource("ex:pred200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(o: new RDFResource("ex:obj200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(l: new RDFPlainLiteral("200")).Count);
Assert.Equal(2, mongoDBStore.SelectQuadruples(c: new RDFContext(new Uri("ex:ctx200")), s: new RDFResource("ex:subj200")).Count);
Assert.Equal(2, mongoDBStore.SelectQuadruples(c: new RDFContext(new Uri("ex:ctx200")), p: new RDFResource("ex:pred200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(c: new RDFContext(new Uri("ex:ctx200")), o: new RDFResource("ex:obj200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(c: new RDFContext(new Uri("ex:ctx200")), l: new RDFPlainLiteral("200")).Count);
Assert.Equal(2, mongoDBStore.SelectQuadruples(c: new RDFContext(new Uri("ex:ctx200")), s: new RDFResource("ex:subj200"), p: new RDFResource("ex:pred200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(c: new RDFContext(new Uri("ex:ctx200")), s: new RDFResource("ex:subj200"), o: new RDFResource("ex:obj200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(c: new RDFContext(new Uri("ex:ctx200")), s: new RDFResource("ex:subj200"), l: new RDFPlainLiteral("200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(c: new RDFContext(new Uri("ex:ctx200")), s: new RDFResource("ex:subj200"), p: new RDFResource("ex:pred200"), o: new RDFResource("ex:obj200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(c: new RDFContext(new Uri("ex:ctx200")), s: new RDFResource("ex:subj200"), p: new RDFResource("ex:pred200"), l: new RDFPlainLiteral("200")).Count);
Assert.Equal(2, mongoDBStore.SelectQuadruples(s: new RDFResource("ex:subj200"), p: new RDFResource("ex:pred200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(s: new RDFResource("ex:subj200"), o: new RDFResource("ex:obj200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(s: new RDFResource("ex:subj200"), l: new RDFPlainLiteral("200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(s: new RDFResource("ex:subj200"), p: new RDFResource("ex:pred200"), o: new RDFResource("ex:obj200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(s: new RDFResource("ex:subj200"), p: new RDFResource("ex:pred200"), l: new RDFPlainLiteral("200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(p: new RDFResource("ex:pred200"), o: new RDFResource("ex:obj200")).Count);
Assert.Equal(1, mongoDBStore.SelectQuadruples(p: new RDFResource("ex:pred200"), l: new RDFPlainLiteral("200")).Count);
Assert.Equal(0, mongoDBStore.SelectQuadruples(p: new RDFResource("ex:pred200786537"), l: new RDFPlainLiteral("200")).Count); //unexisting

//Query (Full Selectors + Accessors)
Assert.True(mongoDBStore.SelectQuadruples().Count > 100);
Assert.Equal(1, mongoDBStore[p: new RDFResource("ex:pred200"), l: new RDFPlainLiteral("200")].QuadruplesCount);
Assert.True(mongoDBStore[null, null, null, null, null].QuadruplesCount > 100);
await mongoDBStore.RemoveQuadruplesAsync();
Assert.Equal(0, mongoDBStore.QuadruplesCount);
Assert.Equal(0, await mongoDBStore.QuadruplesCountAsync);
Assert.True(mongoDBStore.SelectQuadruples().Count == 0);
Assert.True(mongoDBStore[null, null, null, null, null].QuadruplesCount == 0);

Assert.True(1==1);
